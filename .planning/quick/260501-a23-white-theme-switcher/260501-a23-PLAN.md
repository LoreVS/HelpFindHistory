## Task 2 — Create useTheme hook

**Files:** `puzzle-fragments/src/hooks/useTheme.js`

**Action:** Create the file with this content:
```js
import { useState, useEffect } from 'react'

const STORAGE_KEY = 'pf-theme'
const DEFAULT_THEME = 'light'

export function useTheme() {
  const [theme, setTheme] = useState(() => {
    return localStorage.getItem(STORAGE_KEY) || DEFAULT_THEME
  })

  useEffect(() => {
    if (theme === 'dark') {
      document.documentElement.setAttribute('data-theme', 'dark')
    } else {
      document.documentElement.removeAttribute('data-theme')
    }
    localStorage.setItem(STORAGE_KEY, theme)
  }, [theme])

  function toggleTheme() {
    setTheme(t => t === 'light' ? 'dark' : 'light')
  }

  return { theme, toggleTheme }
}
```

**Verify:** File exists at `src/hooks/useTheme.js`.

**Done:** Hook reads localStorage (defaults to 'light'), applies `data-theme` attribute to `<html>`, persists on change.
