```javascript
/** POST /api/attempts/:id/approve — approve attempt, auto-closes project (SCORE-02) */
async approveAttempt(attemptId) {
  const res = await fetch(`${API}/api/attempts/${attemptId}/approve`, {
    method: 'POST',
    headers: authHeaders(),
  })
  if (!res.ok) {
    const body = await res.json().catch(() => ({}))
    throw new Error(body.error || `HTTP ${res.status}`)
  }
  const updated = await res.json()
  set((state) => {
    if (!state.currentProject) return {}
    return {
      currentProject: {
        ...state.currentProject,
        status: 'closed',
        attempts: state.currentProject.attempts.map((a) =>
          a.id === attemptId ? { ...a, status: 'approved' } : a
        ),
      },
    }
  })
  return updated
},

/** POST /api/attempts/:id/reject — reject attempt, canvas unlocks (SCORE-03) */
async rejectAttempt(attemptId) {
  const res = await fetch(`${API}/api/attempts/${attemptId}/reject`, {
    method: 'POST',
    headers: authHeaders(),
  })
  if (!res.ok) {
    const body = await res.json().catch(() => ({}))
    throw new Error(body.error || `HTTP ${res.status}`)
  }
  const updated = await res.json()
  set((state) => {
    if (!state.currentProject) return {}
    return {
      currentProject: {
        ...state.currentProject,
        attempts: state.currentProject.attempts.map((a) =>
          a.id === attemptId ? { ...a, status: 'rejected' } : a
        ),
      },
    }
  })
  return updated
},

/** GET /api/users/me — fetch authenticated user's score and sync to authStore (SCORE-05) */
async fetchUserScore() {
  try {
    const res = await fetch(`${API}/api/users/me`, { headers: authHeaders() })
    if (!res.ok) return
    const data = await res.json()
    useAuthStore.getState().setScore(data.score)
  } catch (err) {
    console.error('[fetchUserScore]', err)
  }
},
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| React component to Zustand store | Components call store actions; store validates nothing — validation is on the server |
| Zustand store to Express API | authHeaders() reads token from store on every call — always fresh (T-03-09 mitigation from Phase 3) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-07 | Elevation of Privilege | approveAttempt / rejectAttempt client calls | accept | Server enforces requireRole('admin') — client-side role checks are cosmetic only. No additional client guard needed here; UI gating in plan 03 is defense-in-depth |
| T-05-08 | Information Disclosure | fetchUserScore stores score in authStore (persisted to localStorage) | accept | score is non-sensitive (integer point total). partialize already persists user object to localStorage; score is part of user object — no new exposure |
| T-05-09 | Tampering | Client could call approveAttempt for any attemptId | accept | Server validates status (published) and role (admin) before any write — client-side ID is untrusted input that the server rejects on invalid state |
</threat_model>
